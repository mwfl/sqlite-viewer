cmake_minimum_required(VERSION 3.21)
project(sqlite_viewer VERSION 0.1.1 LANGUAGES CXX)

include(FetchContent)
option(SQLITE_VIEWER_USE_LOCAL_MWFL "Use a local mwfl checkout" OFF)
set(SQLITE_VIEWER_MWFL_SOURCE_DIR "../mwfl" CACHE PATH "Local mwfl source directory")
set(MWFL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(SQLITE_VIEWER_USE_LOCAL_MWFL)
    add_subdirectory("${SQLITE_VIEWER_MWFL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/mwfl")
else()
  FetchContent_Declare(mwfl GIT_REPOSITORY https://github.com/mwfl/mwfl.git GIT_TAG v0.1.0)
    FetchContent_MakeAvailable(mwfl)
endif()
add_executable(sqlite-viewer WIN32
    main.cpp sqlite_database.cpp sqlite_database.h
    sqlite_viewer.rc resource.h app-icon.ico app.manifest)
target_link_libraries(sqlite-viewer PRIVATE mwfl::app winsqlite3)
target_compile_features(sqlite-viewer PRIVATE cxx_std_20)
target_compile_definitions(sqlite-viewer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX MWFL_APP_VERSION=L\"${PROJECT_VERSION}\")
if(MSVC)
    target_compile_options(sqlite-viewer PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()

enable_testing()
add_executable(sqlite-viewer-tests sqlite_database_test.cpp sqlite_database.cpp sqlite_database.h)
target_link_libraries(sqlite-viewer-tests PRIVATE mwfl::ui winsqlite3)
target_compile_features(sqlite-viewer-tests PRIVATE cxx_std_20)
target_compile_definitions(sqlite-viewer-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(sqlite-viewer-tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()
add_test(NAME sqlite-viewer.model COMMAND sqlite-viewer-tests)
add_test(NAME sqlite-viewer.gui COMMAND sqlite-viewer --self-test)
set_tests_properties(sqlite-viewer.gui PROPERTIES TIMEOUT 20)

install(TARGETS sqlite-viewer RUNTIME DESTINATION . COMPONENT Application)
install(FILES README.md LICENSE DESTINATION . COMPONENT Application)
